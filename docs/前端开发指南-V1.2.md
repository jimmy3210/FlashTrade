## 🔑 核心概念

### 1. 认证流程

```typescript
// 1. 用户登录
POST /auth/login
{
  "username": "trader01",
  "password": "password123"
}

// 2. 获得 Token
{
  "token": "eyJhbGc...",
  "username": "trader01",
  "role": "ROLE_TRADER"
}

// 3. 保存到 localStorage
localStorage.setItem('auth_token', token);

// 4. 后续所有请求自动添加 Authorization 头
Authorization: Bearer {token}
```

### 2. HTTP 客户端 (request.ts)

```typescript
import request from '@/utils/request';

// GET 请求
const data = await request.get<ResponseType>('/endpoint');

// POST 请求
const data = await request.post<ResponseType>('/endpoint', {
  field1: 'value1'
});

// PUT 请求
await request.put<void>('/endpoint/{id}', { ...updates });

// DELETE 请求
await request.delete<void>('/endpoint/{id}');
```

**自动特性**:
- ✅ JSON 序列化/反序列化
- ✅ 自动 Token 注入
- ✅ ApiResponse 包装自动解包
- ✅ 错误统一处理

### 3. 路由导航

### ⭐ UserBrokerBinding (用户账户绑定管理) - 新增

```vue
<template>
  <div class="page">
    <h2>用户-账户绑定管理</h2>

    <div class="section">
      <label>选择用户：</label>
      <select v-model.number="selectedUserId" @change="loadUserBinding">
        <option :value="null">-- 请选择用户 --</option>
        <option v-for="user in users" :key="user.id" :value="user.id">
          {{ user.username }}
        </option>
      </select>
    </div>

    <div v-if="selectedUserId && currentBinding" class="section">
      <h3>当前绑定的账户</h3>
      <div class="binding-info">
        <p><strong>用户：</strong> {{ currentBinding.username }}</p>
        <p><strong>券商：</strong> {{ currentBinding.brokerName }}</p>
        <p><strong>账户号：</strong> {{ currentBinding.accountNo }}</p>
        <p>
          <strong>API 环境：</strong>
          <span class="badge" :class="currentBinding.apiEnvironment">
            {{ currentBinding.apiEnvironment }}
          </span>
        </p>
      </div>
    </div>

    <div v-if="selectedUserId" class="section">
      <h3>切换为其他账户</h3>
      <table v-if="availableAccounts.length > 0">
        <thead>
          <tr>
            <th>操作</th>
            <th>券商</th>
            <th>账户号</th>
            <th>API 环境</th>
          </tr>
        </thead>
        <tbody>
          <tr v-for="account in availableAccounts" :key="account.id">
            <td>
              <button 
                v-if="account.id !== currentBinding?.id"
                @click="bindAccount(account)"
              >
                切换
              </button>
              <span v-else class="current">当前</span>
            </td>
            <td>{{ account.brokerName }}</td>
            <td>{{ account.accountNo }}</td>
            <td>
              <span class="badge" :class="account.apiEnvironment">
                {{ account.apiEnvironment }}
              </span>
            </td>
          </tr>
        </tbody>
      </table>
    </div>

    <div v-if="selectedUserId && !currentBinding" class="warning">
      <p>该用户未绑定任何账户</p>
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref, computed, onMounted } from 'vue';
import request from '@/utils/request';

interface User {
  id: number;
  username: string;
}

interface BrokerAccount {
  id: number;
  username: string;
  brokerName: string;
  accountNo: string;
  apiEnvironment: 'LIVE' | 'SANDBOX';
  status: string;
}

const selectedUserId = ref<number | null>(null);
const users = ref<User[]>([]);
const allAccounts = ref<BrokerAccount[]>([]);
const currentBinding = ref<BrokerAccount | null>(null);

const availableAccounts = computed(() => {
  return allAccounts.value.filter(a => a.status === 'ACTIVE');
});

const loadUsers = async () => {
  users.value = await request.get<User[]>('/admin/users');
};

const loadUserBinding = async () => {
  if (!selectedUserId.value) {
    currentBinding.value = null;
    return;
  }

  try {
    currentBinding.value = await request.get<BrokerAccount>(
      `/admin/broker-accounts/user/${selectedUserId.value}/current`
    );
  } catch {
    currentBinding.value = null;
  }
};

const bindAccount = async (account: BrokerAccount) => {
  if (!selectedUserId.value) return;

  const username = users.value.find(u => u.id === selectedUserId.value)?.username;
  if (!confirm(`确认将用户 ${username} 绑定到账户 ${account.accountNo}?`)) {
    return;
  }

  try {
    await request.post(
      `/admin/broker-accounts/user/${selectedUserId.value}/bind/${account.id}`,
      {}
    );
    alert('绑定成功');
    await loadUserBinding();
  } catch (error) {
    alert(`绑定失败: ${(error as any).message}`);
  }
};

onMounted(async () => {
  await loadUsers();
  allAccounts.value = await request.get<BrokerAccount[]>('/admin/broker-accounts');
});
</script>
```

### 状态管理 (Pinia)

```typescript
// stores/auth.ts
import { defineStore } from 'pinia';
import { ref } from 'vue';

export const useAuthStore = defineStore('auth', () => {
  const token = ref<string | null>(localStorage.getItem('auth_token'));
  const username = ref<string | null>(localStorage.getItem('username'));

  const logout = () => {
    token.value = null;
    username.value = null;
    localStorage.removeItem('auth_token');
    localStorage.removeItem('username');
  };

  return { token, username, logout };
});

// 在组件中使用
import { useAuthStore } from '@/stores/auth';

const authStore = useAuthStore();
console.log(authStore.username);
authStore.logout();
```

## 📱 交易客户端集成指南

### 关键新增：查询当前账户 API

```typescript
import request from '@/utils/request';

interface CurrentAccount {
  id: number;
  accountNo: string;
  brokerName: string;
  apiEnvironment: 'LIVE' | 'SANDBOX';
  status: 'ACTIVE' | 'SUSPENDED' | 'CLOSED';
}

// 在用户登录或切换账户时调用
const getCurrentAccount = async (): Promise<CurrentAccount> => {
  return await request.get<CurrentAccount>('/api/v1/trade/current-account');
};
```

### 集成示例

```typescript
// 1. 用户登录完成
const handleLogin = async (token: string) => {
  localStorage.setItem('auth_token', token);
  
  // 2. 立即查询用户的当前账户
  try {
    const account = await getCurrentAccount();
    
    // 3. 根据 apiEnvironment 切换端点
    const brokerClient = createBrokerClient({
      broker: account.brokerName,
      environment: account.apiEnvironment,  // LIVE 或 SANDBOX
      accountNumber: account.accountNo
    });
    
    // 4. 初始化交易界面
    initializeTradeUI(account);
  } catch (error) {
    console.error('无法获取账户信息:', error);
    // 提示用户联系管理员
  }
};

// 5. 定期检查（如用户切换账户时）
const checkAccountChange = async () => {
  const previousAccount = currentAccount.value;
  const newAccount = await getCurrentAccount();
  
  if (previousAccount.id !== newAccount.id) {
    // 账户已切换
    console.log(`账户从 ${previousAccount.accountNo} 切换到 ${newAccount.accountNo}`);
    
    // 重新初始化客户端
    reinitializeBrokerClient(newAccount);
  }
};
```

## 🐛 常见问题

### Q: Token 过期如何处理？
**A**: 在 `request.ts` 中的响应拦截器中检查 401 错误，自动重定向到登录页。

### Q: 如何在多个页面间共享数据？
**A**: 使用 Pinia store:
```typescript
// stores/account.ts
export const useAccountStore = defineStore('account', () => {
  const currentAccount = ref<BrokerAccount | null>(null);
  // ...
});
```

### Q: CSS 污染问题？
**A**: 始终在 `<style scoped>` 中编写样式。

### Q: 如何处理异步操作的加载状态？
**A**: 
```typescript
const isLoading = ref(false);

const loadData = async () => {
  isLoading.value = true;
  try {
    data.value = await request.get('/api/data');
  } finally {
    isLoading.value = false;
  }
};
```
